```cpp
// Thread-safe publish method with retry logic
bool publishMessage(const char* topic, const char* payload, bool retained = false) {
    if (!isConnected() || !config.enabled) {
        return false;
    }

    // Add to publish queue if not connected
    if (currentState != MQTT_CONNECTED) {
        return queueMessage(topic, payload, retained);
    }

    // Attempt publish with retries
    uint8_t retries = 3;
    while (retries > 0) {
        if (mqttClient.publish(topic, payload, retained)) {
            updateStats(STAT_MESSAGES_SENT);
            return true;
        }
        retries--;
        delay(100);  // Brief delay between retries
    }

    updateStats(STAT_MESSAGES_FAILED);
    return false;
}
```

```cpp
void updateStateMachine() {
    static uint32_t stateTimer = 0;

    switch(currentState) {
        case MQTT_DISCONNECTED:
            if (config.enabled && WiFi.status() == WL_CONNECTED) {
                if (millis() - reconnectParams.lastAttemptTime >= reconnectParams.currentDelay) {
                    changeState(MQTT_CONNECTING);
                    attemptConnection();
                }
            }
            break;

        case MQTT_CONNECTING:
            if (millis() - stateTimer > 5000) {  // 5 second timeout
                handleConnectionTimeout();
                changeState(MQTT_ERROR);
            }
            break;

        case MQTT_CONNECTED:
            if (!mqttClient.connected()) {
                handleDisconnection();
                changeState(MQTT_DISCONNECTED);
            }
            break;

        case MQTT_ERROR:
            updateReconnectionDelay();
            changeState(MQTT_DISCONNECTED);
            break;
    }
}
```

```cpp
// Pre-allocated buffers to avoid fragmentation
class MQTTBufferPool {
private:
    static const size_t BUFFER_SIZE = 256;
    static const size_t POOL_SIZE = 5;

    struct Buffer {
        char data[BUFFER_SIZE];
        bool inUse;
    };

    Buffer pool[POOL_SIZE];

public:
    char* acquire() {
        for (auto& buffer : pool) {
            if (!buffer.inUse) {
                buffer.inUse = true;
                return buffer.data;
            }
        }
        return nullptr;  // Pool exhausted
    }

    void release(char* ptr) {
        for (auto& buffer : pool) {
            if (buffer.data == ptr) {
                buffer.inUse = false;
                memset(buffer.data, 0, BUFFER_SIZE);
                break;
            }
        }
    }
};

// Memory monitoring
struct MemoryStats {
    uint32_t heapAtStart;
    uint32_t currentHeap;
    uint32_t minFreeHeap;
    uint32_t maxAllocated;

    void update() {
        currentHeap = ESP.getFreeHeap();
        if (currentHeap < minFreeHeap) {
            minFreeHeap = currentHeap;
        }
        uint32_t allocated = heapAtStart - currentHeap;
        if (allocated > maxAllocated) {
            maxAllocated = allocated;
        }
    }

    bool isWithinLimits() {
        return (heapAtStart - currentHeap) < 30720;  // 30KB limit
    }
};
```

```cpp
// Non-blocking connection with timeout
bool attemptConnectionNonBlocking() {
    static enum {
        CONN_IDLE,
        CONN_STARTED,
        CONN_WAITING,
        CONN_SUCCESS,
        CONN_FAILED
    } connState = CONN_IDLE;

    static uint32_t connStartTime = 0;
    const uint32_t CONN_TIMEOUT = 5000;  // 5 second timeout

    switch(connState) {
        case CONN_IDLE:
            connState = CONN_STARTED;
            connStartTime = millis();
            // Fall through

        case CONN_STARTED:
            if (mqttClient.connect(config.clientId, config.username, config.password)) {
                connState = CONN_SUCCESS;
                return true;
            }
            connState = CONN_WAITING;
            break;

        case CONN_WAITING:
            if (millis() - connStartTime > CONN_TIMEOUT) {
                connState = CONN_FAILED;
                return false;
            }
            // Check again
            if (mqttClient.connected()) {
                connState = CONN_SUCCESS;
                return true;
            }
            break;

        case CONN_SUCCESS:
        case CONN_FAILED:
            connState = CONN_IDLE;  // Reset for next attempt
            break;
    }

    return false;  // Still in progress
}

// Batch message publishing for efficiency
class MessageBatcher {
private:
    static const size_t MAX_BATCH_SIZE = 10;
    struct Message {
        char topic[64];
        char payload[256];
        bool retained;
    };
    Message batch[MAX_BATCH_SIZE];
    size_t batchCount = 0;
    uint32_t lastFlush = 0;

public:
    void add(const char* topic, const char* payload, bool retained = false) {
        if (batchCount < MAX_BATCH_SIZE) {
            strncpy(batch[batchCount].topic, topic, sizeof(batch[batchCount].topic));
            strncpy(batch[batchCount].payload, payload, sizeof(batch[batchCount].payload));
            batch[batchCount].retained = retained;
            batchCount++;
        }
    }

    void flush() {
        if (batchCount == 0) return;

        for (size_t i = 0; i < batchCount; i++) {
            MQTTManager::getInstance()->publishMessage(
                batch[i].topic,
                batch[i].payload,
                batch[i].retained
            );
        }
        batchCount = 0;
        lastFlush = millis();
    }

    bool shouldFlush() {
        return batchCount >= MAX_BATCH_SIZE ||
               (batchCount > 0 && millis() - lastFlush > 1000);
    }
};
```

```cpp
// Handle configuration changes during runtime
void handleConfigChange(const MQTTConfig& oldConfig, const MQTTConfig& newConfig) {
    bool needsReconnect = false;

    // Check what changed
    if (strcmp(oldConfig.broker, newConfig.broker) != 0) {
        logEvent("Broker address changed", LOG_INFO);
        needsReconnect = true;
    }

    if (oldConfig.port != newConfig.port) {
        logEvent("Port changed", LOG_INFO);
        needsReconnect = true;
    }

    if (strcmp(oldConfig.username, newConfig.username) != 0 ||
        strcmp(oldConfig.password, newConfig.password) != 0) {
        logEvent("Credentials changed", LOG_INFO);
        needsReconnect = true;
    }

    if (oldConfig.enabled != newConfig.enabled) {
        if (newConfig.enabled) {
            logEvent("MQTT enabled", LOG_INFO);
            connect();
        } else {
            logEvent("MQTT disabled", LOG_INFO);
            disconnect();
        }
        return;
    }

    if (needsReconnect && isConnected()) {
        logEvent("Configuration requires reconnection", LOG_INFO);
        disconnect();
        connect();
    }
}
```

```cpp
// Prevent rapid reconnection attempts
bool shouldAttemptReconnect() {
    uint32_t now = millis();

    // Check minimum delay between attempts
    if (now - reconnectParams.lastAttemptTime < reconnectParams.currentDelay) {
        return false;
    }

    // Check maximum attempts per hour
    static uint8_t attemptsThisHour = 0;
    static uint32_t hourStart = 0;

    if (now - hourStart > 3600000) {  // New hour
        attemptsThisHour = 0;
        hourStart = now;
    }

    if (attemptsThisHour >= MAX_ATTEMPTS_PER_HOUR) {
        logEvent("Maximum reconnection attempts reached for this hour", LOG_WARNING);
        return false;
    }

    attemptsThisHour++;
    return true;
}

// Handle concurrent operations
class MQTTMutex {
private:
    volatile bool locked = false;
    uint32_t lockTimeout = 1000;  // 1 second timeout
    uint32_t lockTime = 0;

public:
    bool acquire(const char* owner) {
        uint32_t startTime = millis();

        while (locked) {
            if (millis() - startTime > lockTimeout) {
                logEvent("Mutex timeout for: " + String(owner), LOG_ERROR);
                return false;
            }
            yield();  // Allow other tasks
        }

        locked = true;
        lockTime = millis();
        return true;
    }

    void release() {
        locked = false;
    }

    bool isLocked() {
        // Auto-release stale locks
        if (locked && millis() - lockTime > lockTimeout * 2) {
            logEvent("Auto-releasing stale lock", LOG_WARNING);
            locked = false;
        }
        return locked;
    }
};
```

```cpp
// Performance monitoring
struct PerformanceMetrics {
    uint32_t connectTime;      // Time to establish connection
    uint32_t publishTime;       // Average publish time
    uint32_t publishCount;      // Number of publishes
    uint32_t maxPublishTime;    // Maximum publish time
    uint32_t minPublishTime;    // Minimum publish time
    uint32_t loopTime;          // Average loop time
    uint32_t maxLoopTime;       // Maximum loop time

    void recordPublish(uint32_t duration) {
        publishTime = ((publishTime * publishCount) + duration) / (publishCount + 1);
        publishCount++;

        if (duration > maxPublishTime || maxPublishTime == 0) {
            maxPublishTime = duration;
        }
        if (duration < minPublishTime || minPublishTime == 0) {
            minPublishTime = duration;
        }
    }

    void printReport() {
        Serial.println("=== Performance Report ===");
        Serial.printf("Connection Time: %u ms\n", connectTime);
        Serial.printf("Publishes: %u\n", publishCount);
        if (publishCount > 0) {
            Serial.printf("  - Average: %u ms\n", publishTime);
            Serial.printf("  - Min: %u ms\n", minPublishTime);
            Serial.printf("  - Max: %u ms\n", maxPublishTime);
        }
        Serial.printf("Loop Time:\n");
        Serial.printf("  - Average: %u µs\n", loopTime);
        Serial.printf("  - Max: %u µs\n", maxLoopTime);
        Serial.println("======================");
    }
};
```

```cpp
// Subscription management
class SubscriptionManager {
private:
    struct Subscription {
        char topic[64];
        uint8_t qos;
        void (*callback)(const char* topic, const char* payload);
        bool active;
    };

    static const size_t MAX_SUBSCRIPTIONS = 10;
    Subscription subscriptions[MAX_SUBSCRIPTIONS];

public:
    bool subscribe(const char* topic, uint8_t qos,
                  void (*callback)(const char*, const char*)) {
        // Find empty slot
        for (auto& sub : subscriptions) {
            if (!sub.active) {
                strncpy(sub.topic, topic, sizeof(sub.topic) - 1);
                sub.qos = qos;
                sub.callback = callback;
                sub.active = true;

                // Subscribe via MQTT client
                if (MQTTManager::getInstance()->isConnected()) {
                    return mqttClient.subscribe(topic, qos);
                }
                return true;  // Will subscribe on connect
            }
        }
        return false;  // No slots available
    }

    void resubscribeAll() {
        for (const auto& sub : subscriptions) {
            if (sub.active) {
                mqttClient.subscribe(sub.topic, sub.qos);
                logEvent("Resubscribed to: " + String(sub.topic), LOG_DEBUG);
            }
        }
    }

    void handleMessage(const char* topic, const char* payload) {
        for (const auto& sub : subscriptions) {
            if (sub.active && topicMatch(sub.topic, topic)) {
                sub.callback(topic, payload);
            }
        }
    }

private:
    bool topicMatch(const char* pattern, const char* topic) {
        // Implement MQTT topic wildcard matching
        // Support + for single level and # for multi-level wildcards
        // This is a simplified version
        return strcmp(pattern, topic) == 0;
    }
};

// Command handling via MQTT
void setupCommandSubscriptions() {
    SubscriptionManager* subMgr = MQTTManager::getInstance()->getSubscriptionManager();

    // Subscribe to command topics
    subMgr->subscribe("tempcontroller/cmd/+", 1, [](const char* topic, const char* payload) {
        // Extract command from topic
        const char* cmd = strrchr(topic, '/');
        if (cmd) {
            cmd++;  // Skip the '/'
            handleCommand(cmd, payload);
        }
    });

    // Subscribe to configuration updates
    subMgr->subscribe("tempcontroller/config/+", 1, [](const char* topic, const char* payload) {
        // Handle configuration updates
        const char* configKey = strrchr(topic, '/');
        if (configKey) {
            configKey++;
            updateConfiguration(configKey, payload);
        }
    });
}
```
