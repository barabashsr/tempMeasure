81void MQTTManager::loop() {
82 if (!mqttClient.connected()) {
83 if (!isConnected) {
84 // First time noticing disconnection
85 Serial.println("[MQTTManager] MQTT connection lost!");
86 isConnected = false;
87 }
88
89 // Check if it's time to reconnect
90 unsigned long now = millis();
91 if (now - lastReconnectAttempt > RECONNECT_INTERVAL) {
92 Serial.println("[MQTTManager] Attempting reconnection...");
93 lastReconnectAttempt = now;
94
95 if (attemptConnection()) {
96 Serial.println("[MQTTManager] Reconnection successful!");
97 lastReconnectAttempt = 0;
98 } else {
99 Serial.print("[MQTTManager] Reconnection failed. Will retry in ");
100 Serial.print(RECONNECT_INTERVAL / 1000);
101 Serial.println(" seconds.");
102 }
103 }
104 } else {
105 // Maintain connection
106 mqttClient.loop();
107
108 // Publish incrementing counter every second
109 unsigned long now = millis();
110 if (now - lastPublishTime >= PUBLISH_INTERVAL) {
111 lastPublishTime = now;
112
113 // Create message with incrementing counter
114 String message = String(publishCounter++);
115
116 Serial.print("[MQTTManager] Publishing counter: ");
117 Serial.println(message);
118
119 if (publish(message)) {
120 Serial.println("[MQTTManager] Counter published successfully");
121 } else {
122 Serial.println("[MQTTManager] Failed to publish counter");
123 }
124 }
125 }

132bool MQTTManager::attemptConnection() {
133 Serial.print("[MQTTManager] Connecting to MQTT broker... ");
134
135 if (mqttClient.connect(DEVICE_ID, MQTT_USERNAME, MQTT_PASSWORD)) {
136 Serial.println("CONNECTED!");
137 isConnected = true;
138
139 // Subscribe to topic
140 Serial.print("[MQTTManager] Subscribing to topic: ");
141 Serial.print(SUBSCRIBE_TOPIC);
142
143 if (mqttClient.subscribe(SUBSCRIBE_TOPIC)) {
144 Serial.println(" - SUCCESS!");
145 } else {
146 Serial.println(" - FAILED!");
147 }
148
149 // Send connection announcement
150 Serial.println("[MQTTManager] Sending connection announcement...");
151 publish("Device connected and ready");
152
153 return true;
154 } else {
155 Serial.print("FAILED! rc=");
156 Serial.print(mqttClient.state());
157 Serial.print(" - ");
158
159 // Decode error codes
160 switch(mqttClient.state()) {
161 case -4: Serial.println("MQTT_CONNECTION_TIMEOUT"); break;
162 case -3: Serial.println("MQTT_CONNECTION_LOST"); break;
163 case -2: Serial.println("MQTT_CONNECT_FAILED"); break;
164 case -1: Serial.println("MQTT_DISCONNECTED"); break;
165 case 0: Serial.println("MQTT_CONNECTED"); break;
166 case 1: Serial.println("MQTT_CONNECT_BAD_PROTOCOL"); break;
167 case 2: Serial.println("MQTT_CONNECT_BAD_CLIENT_ID"); break;
168 case 3: Serial.println("MQTT_CONNECT_UNAVAILABLE"); break;
169 case 4: Serial.println("MQTT_CONNECT_BAD_CREDENTIALS"); break;
170 case 5: Serial.println("MQTT_CONNECT_UNAUTHORIZED"); break;
171 default: Serial.println("UNKNOWN ERROR"); break;
172 }
173
174 isConnected = false;
175 return false;
176 }
